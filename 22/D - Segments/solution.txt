import sys
 
def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
 
    n = int(input_data[0])
    segments = []
    idx = 1
    for _ in range(n):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        segments.append((min(u, v), max(u, v)))
 
    # Sort segments by right endpoint
    segments.sort(key=lambda x: x[1])
 
    nails = []
    last_nail = -float('inf')
 
    for l, r in segments:
        if last_nail < l:
            last_nail = r
            nails.append(r)
 
    print(len(nails))
    print(*(nails))
 
if __name__ == '__main__':
    solve()
