import sys
 
def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
 
    n = int(input_data[0])
    m = int(input_data[1])
    v = int(input_data[2])
 
    min_m = n - 1
    max_m = 1 + (n - 1) * (n - 2) // 2
 
    if m < min_m or m > max_m:
        print(-1)
        return
 
    # Select an isolated vertex u != v
    u = 2 if v == 1 else 1
 
    # Remaining vertices
    rest = [i for i in range(1, n + 1) if i != v and i != u]
 
    out = []
    edges_added = 0
 
   
    out.append(f"{u} {v}")
    edges_added += 1
 
 
    out.append(f"{v} {rest[0]}")
    edges_added += 1
    for i in range(len(rest) - 1):
        out.append(f"{rest[i]} {rest[i + 1]}")
        edges_added += 1
 
 
    for i in range(1, len(rest)):
        if edges_added >= m:
            break
        out.append(f"{v} {rest[i]}")
        edges_added += 1
 
    if edges_added < m:
        for i in range(len(rest)):
            for j in range(i + 2, len(rest)):
                if edges_added >= m:
                    break
                out.append(f"{rest[i]} {rest[j]}")
                edges_added += 1
            if edges_added >= m:
                break
 
    sys.stdout.write("
".join(out) + "
")
 
if __name__ == '__main__':
    solve()
