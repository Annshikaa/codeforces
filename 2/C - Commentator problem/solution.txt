import math
import sys
 
x1, y1, r1 = map(float, sys.stdin.readline().split())
x2, y2, r2 = map(float, sys.stdin.readline().split())
x3, y3, r3 = map(float, sys.stdin.readline().split())
 
A1 = 2 * (x1 - x2)
B1 = 2 * (y1 - y2)
C1 = x1 * x1 + y1 * y1 - x2 * x2 - y2 * y2
D1 = r2 * r2 - r1 * r1
 
A2 = 2 * (x1 - x3)
B2 = 2 * (y1 - y3)
C2 = x1 * x1 + y1 * y1 - x3 * x3 - y3 * y3
D2 = r3 * r3 - r1 * r1
 
det = A1 * B2 - A2 * B1
 
# x = px * t + qx
# y = py * t + qy
px = (D1 * B2 - D2 * B1) / det
qx = (C1 * B2 - C2 * B1) / det
 
py = (A1 * D2 - A2 * D1) / det
qy = (A1 * C2 - A2 * C1) / det
 
dx = qx - x1
dy = qy - y1
 
a = px * px + py * py
b = 2 * (px * dx + py * dy) - r1 * r1
c = dx * dx + dy * dy
 
eps = 1e-12
roots = []
 
if abs(a) < eps:
    if abs(b) > eps:
        roots.append(-c / b)
else:
    disc = b * b - 4 * a * c
    if disc >= -eps:
        disc = max(0.0, disc)
        s = math.sqrt(disc)
        roots.append((-b - s) / (2 * a))
        roots.append((-b + s) / (2 * a))
 
best_t = None
for t in roots:
    if t > eps:
        if best_t is None or t < best_t:
            best_t = t
 
if best_t is not None:
    x = px * best_t + qx
    y = py * best_t + qy
    print(f"{x:.5f} {y:.5f}")
