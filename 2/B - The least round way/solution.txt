import sys
input = sys.stdin.buffer.readline
 
n = int(input())
INF = 10 ** 9
 
a2 = [[0] * n for _ in range(n)]
a5 = [[0] * n for _ in range(n)]
 
zx = zy = -1
 
for i in range(n):
    row = list(map(int, input().split()))
    for j in range(n):
        x = row[j]
 
        if x == 0:
            zx, zy = i, j
            a2[i][j] = a5[i][j] = 1
            continue
 
        t = x
        while (t & 1) == 0:
            a2[i][j] += 1
            t >>= 1
 
        while x % 5 == 0:
            a5[i][j] += 1
            x //= 5
 
def solve(cost):
    dp = [[INF] * n for _ in range(n)]
    par = [[''] * n for _ in range(n)]
 
    dp[0][0] = cost[0][0]
 
    for i in range(n):
        for j in range(n):
            if i == 0 and j == 0:
                continue
 
            if i > 0 and dp[i - 1][j] + cost[i][j] < dp[i][j]:
                dp[i][j] = dp[i - 1][j] + cost[i][j]
                par[i][j] = 'D'
 
            if j > 0 and dp[i][j - 1] + cost[i][j] < dp[i][j]:
                dp[i][j] = dp[i][j - 1] + cost[i][j]
                par[i][j] = 'R'
 
    path = []
    i, j = n - 1, n - 1
 
    while i > 0 or j > 0:
        path.append(par[i][j])
        if par[i][j] == 'D':
            i -= 1
        else:
            j -= 1
 
    path.reverse()
    return dp[n - 1][n - 1], ''.join(path)
 
v2, p2 = solve(a2)
v5, p5 = solve(a5)
 
best = min(v2, v5)
 
if zx != -1 and best > 1:
    print(1)
    print('D' * zx + 'R' * zy + 'D' * (n - 1 - zx) + 'R' * (n - 1 - zy))
else:
    if v2 < v5:
        print(v2)
        print(p2)
    else:
        print(v5)
        print(p5)
