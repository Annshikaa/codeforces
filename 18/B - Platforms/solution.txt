import sys
 
def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    d = int(input_data[1])
    m = int(input_data[2])
    l = int(input_data[3])
 
    # Check gaps between consecutive platforms
    for k in range(1, n):
        plat_end = (k - 1) * m + l
        next_plat_start = k * m
        
        # First jump landing strictly after plat_end
        jump_coord = (plat_end // d + 1) * d
        
        # If it falls in the gap before the next platform starts
        if jump_coord < next_plat_start:
            print(jump_coord)
            return
 
    # If Bob successfully passes all gaps, he falls after the last platform
    last_end = (n - 1) * m + l
    jump_coord = (last_end // d + 1) * d
    print(jump_coord)
 
if __name__ == '__main__':
    solve()
