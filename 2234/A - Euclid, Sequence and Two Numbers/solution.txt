import sys
 
input = sys.stdin.readline
 
t = int(input())
 
for _ in range(t):
    n = int(input())
    b = list(map(int, input().split()))
 
    b.sort(reverse=True)
 
    if n == 2:
        print(b[0], b[1])
        continue
 
    ok = True
 
    for i in range(n - 2):
        if b[i] % b[i + 1] != b[i + 2]:
            ok = False
            break
 
    if ok:
        print(b[0], b[1])
    else:
        print(-1)
