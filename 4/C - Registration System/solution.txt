import sys
 
def solve():
    # Read all inputs from standard input for fast I/O
    input_data = sys.stdin.read().splitlines()
    if not input_data:
        return
    
    n = int(input_data[0])
    database = {}
    output = []
    
    for i in range(1, n + 1):
        name = input_data[i]
        
        if name not in database:
            # First time seeing this name
            database[name] = 1
            output.append("OK")
        else:
            # Name already exists, append the current count
            count = database[name]
            new_name = f"{name}{count}"
            output.append(new_name)
            
            # Increment the count for the base name
            database[name] += 1
            
    # Print all results at once
    print('
'.join(output))
 
if __name__ == '__main__':
    solve()
