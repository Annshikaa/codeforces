import sys
 
def solve():
    # Read all inputs from standard input efficiently
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    d = int(input_data[1])
    b = [int(x) for x in input_data[2:]]
    
    total_moves = 0
    
    # Iterate through the sequence starting from the second element
    for i in range(1, n):
        if b[i] <= b[i-1]:
            # Calculate the minimum number of additions of 'd' required
            diff = b[i-1] - b[i]
            count = (diff // d) + 1
            
            # Update the total moves and the current element
            total_moves += count
            b[i] += count * d
            
    print(total_moves)
 
if __name__ == '__main__':
    solve()
