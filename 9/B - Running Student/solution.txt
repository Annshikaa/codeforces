import sys
import math
 
def main():
    # Read all inputs from standard input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
        
    n = int(input_data[0])
    vb = float(input_data[1])
    vs = float(input_data[2])
    
    # Coordinates of the bus stops
    x = [float(val) for val in input_data[3:3+n]]
    
    # Coordinates of the University
    xu = float(input_data[3+n])
    yu = float(input_data[3+n+1])
    
    best_index = -1
    min_time = float('inf')
    min_dist_to_uni = float('inf')
    
    eps = 1e-9  # Epsilon for safe floating-point comparison
    
    # Student cannot get off at the first stop, so we start at index 1 (2nd stop)
    for i in range(1, n):
        # Distance from the current bus stop (x[i], 0) to University (xu, yu)
        dist_to_uni = math.hypot(x[i] - xu, yu)
        
        time_bus = x[i] / vb
        time_run = dist_to_uni / vs
        total_time = time_bus + time_run
        
        # Check if we found a strictly faster route
        if total_time < min_time - eps:
            min_time = total_time
            min_dist_to_uni = dist_to_uni
            best_index = i + 1  # 1-based indexing
            
        # If times are practically identical, break ties by distance to University
        elif abs(total_time - min_time) <= eps:
            if dist_to_uni < min_dist_to_uni - eps:
                min_dist_to_uni = dist_to_uni
                best_index = i + 1
                
    print(best_index)
 
if __name__ == '__main__':
    main()
