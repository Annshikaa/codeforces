import sys
 
def can_see_sequences(text, a, b):
    # Find the earliest occurrence of the first sequence 'a'
    pos_a = text.find(a)
    
    # If 'a' is not found, Peter couldn't have seen it
    if pos_a == -1:
        return False
        
    # Find the sequence 'b' starting strictly after 'a' ends
    start_search_for_b = pos_a + len(a)
    pos_b = text.find(b, start_search_for_b)
    
    # Return True if 'b' is found after 'a'
    return pos_b != -1
 
def main():
    # Read all lines from standard input
    input_data = sys.stdin.read().split()
    if len(input_data) < 3:
        return
        
    s = input_data[0]
    a = input_data[1]
    b = input_data[2]
    
    # Check forward journey
    forward_possible = can_see_sequences(s, a, b)
    
    # Check backward journey (reverse the stations string)
    backward_possible = can_see_sequences(s[::-1], a, b)
    
    # Print the result based on the conditions met
    if forward_possible and backward_possible:
        print("both")
    elif forward_possible:
        print("forward")
    elif backward_possible:
        print("backward")
    else:
        print("fantasy")
 
if __name__ == '__main__':
    main()
