import sys
 
MOD = 10**9 + 7
 
 
def solve_case(n, b):
    # ------------------------------------------------------------
    # 1. Mark every position that cannot contain treasure.
    # ------------------------------------------------------------
    diff = [0] * (n + 1)
    all_unknown = True
 
    for i, d in enumerate(b):
        if d != -1:
            all_unknown = False
 
        if d > 0:
            # Every position whose distance from i is < d is forbidden.
            left = max(0, i - d + 1)
            right = min(n - 1, i + d - 1)
 
            if left <= right:
                diff[left] += 1
                diff[right + 1] -= 1
 
    forbidden = [False] * n
 
    cur = 0
    for i in range(n):
        cur += diff[i]
        forbidden[i] = (cur > 0)
 
    # ------------------------------------------------------------
    # 2. Compress all positions that are still allowed to contain
    #    treasure.
    # ------------------------------------------------------------
    rank = [-1] * n
    m = 0
 
    for i in range(n):
        if not forbidden[i]:
            rank[i] = m
            m += 1
 
    if m == 0:
        return 0
 
    # forced[k] = this surviving position MUST contain treasure
    forced = [False] * m
 
    # edge[k] = at least one of positions k and k+1 must contain treasure
    edge = [False] * max(0, m - 1)
 
    # ------------------------------------------------------------
    # 3. Convert every known b[i] into forced positions / OR edges.
    # ------------------------------------------------------------
    for i, d in enumerate(b):
 
        if d == 0:
            # Distance 0 means island i itself has treasure.
            if forbidden[i]:
                return 0
 
            forced[rank[i]] = True
 
        elif d > 0:
            candidates = []
 
            left = i - d
            right = i + d
 
            if left >= 0 and not forbidden[left]:
                candidates.append(rank[left])
 
            if right < n and not forbidden[right]:
                candidates.append(rank[right])
 
            # No possible island exactly d away.
            if not candidates:
                return 0
 
            if len(candidates) == 1:
                # Only one endpoint can contain treasure,
                # so it is forced.
                forced[candidates[0]] = True
 
            else:
                a, c = candidates
 
                if a > c:
                    a, c = c, a
 
                # Everything between these endpoints was forbidden,
                # therefore they must be consecutive surviving positions.
                if c != a + 1:
                    return 0
 
                edge[a] = True
 
    # ------------------------------------------------------------
    # 4. DP over surviving positions.
    #
    # dp0 = number of assignments where current position has no treasure
    # dp1 = number of assignments where current position has treasure
    # ------------------------------------------------------------
    if forced[0]:
        dp0 = 0
        dp1 = 1
    else:
        dp0 = 1
        dp1 = 1
 
    for i in range(1, m):
        total = (dp0 + dp1) % MOD
 
        if edge[i - 1]:
            # Previous OR current must be 1.
            #
            # current = 0 => previous MUST be 1
            ndp0 = dp1
            #
            # current = 1 => previous can be anything
            ndp1 = total
        else:
            # No constraint between them.
            ndp0 = total
            ndp1 = total
 
        if forced[i]:
            ndp0 = 0
 
        dp0 = ndp0 % MOD
        dp1 = ndp1 % MOD
 
    answer = (dp0 + dp1) % MOD
 
    # If every element was destroyed, DP also counted the empty set.
    # But Bessie buried treasure on at least one island.
    if all_unknown:
        answer = (answer - 1) % MOD
 
    return answer
 
 
def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)
 
    t = next(it)
 
    treasure_map_fin = []
 
    for _ in range(t):
        n = next(it)
        b = [next(it) for _ in range(n)]
 
        treasure_map_fin.append(str(solve_case(n, b)))
 
    sys.stdout.write("
".join(treasure_map_fin))
 
 
if __name__ == "__main__":
    solve()
