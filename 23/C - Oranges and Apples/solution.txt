import sys
 
def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
 
    it = iter(input_data)
    t = int(next(it))
    out = []
 
    for _ in range(t):
        n = int(next(it))
        m = 2 * n - 1
        
        boxes = []
        total_o = 0
        for i in range(1, m + 1):
            a = int(next(it))
            o = int(next(it))
            boxes.append((a, o, i))
            total_o += o
 
        # Sort by apple count
        boxes.sort(key=lambda x: x[0])
 
        # Check even indices sum of oranges
        even_o = sum(boxes[i][1] for i in range(0, m, 2))
 
        out.append("YES")
        if even_o * 2 >= total_o:
            res = [str(boxes[i][2]) for i in range(0, m, 2)]
        else:
            res = [str(boxes[i][2]) for i in range(1, m, 2)] + [str(boxes[-1][2])]
 
        out.append(" ".join(res))
 
    sys.stdout.write("
".join(out) + "
")
 
if __name__ == "__main__":
    main()
