import sys
 
def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    t = int(input_data[0])
    idx = 1
    
    out = []
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        idx += 2
        
        if x % 2 == 0 or y % 2 == 0:
            out.append("YES")
        else:
            out.append("NO")
            
    print('
'.join(out))
 
if __name__ == '__main__':
    main()
