import sys
 
def read_segments():
    lines = sys.stdin.read().split()
    if not lines:
        return []
    
    t = int(lines[0])
    idx = 1
    test_cases = []
    for _ in range(t):
        case = []
        for _ in range(3):
            x1, y1, x2, y2 = map(int, lines[idx:idx+4])
            case.append(((x1, y1), (x2, y2)))
            idx += 4
        test_cases.append(case)
    return test_cases
 
def cross_product(p1, p2, p3):
    # Vector p1->p2 X p1->p3
    return (p2[0] - p1[0]) * (p3[1] - p1[1]) - (p2[1] - p1[1]) * (p3[0] - p1[0])
 
def dot_product(p1, p2, p3):
    # Vector p1->p2 . p1->p3
    return (p2[0] - p1[0]) * (p3[0] - p1[0]) + (p2[1] - p1[1]) * (p3[1] - p1[1])
 
def dist_sq(p1, p2):
    return (p1[0] - p2[0])**2 + (p1[1] - p2[1])**2
 
def is_on_segment_with_ratio(a, b, p):
    """Checks if P lies on segment AB and divides it in ratio >= 1/4."""
    # Must be collinear
    if cross_product(a, b, p) != 0:
        return False
    
    # Must lie within the bounding box of AB
    if min(a[0], b[0]) <= p[0] <= max(a[0], b[0]) and min(a[1], b[1]) <= p[1] <= max(a[1], b[1]):
        # Calculate squared distances
        d_ap = dist_sq(a, p)
        d_pb = dist_sq(p, b)
        
        # Avoid checking ratio if P is at one of the endpoints
        if d_ap == 0 or d_pb == 0:
            return False
        
        # Min len / Max len >= 1/4  =>  Min len^2 / Max len^2 >= 1/16
        return 16 * min(d_ap, d_pb) >= max(d_ap, d_pb)
        
    return False
 
def solve_case(segments):
    # Try all 3 choices for which segment is the bridge
    for i in range(3):
        bridge = segments[i]
        legs = [segments[j] for j in range(3) if j != i]
        
        # Try all 4 combinations of leg orientations to find a common vertex
        for leg1_start, leg1_end in [legs[0], legs[0][::-1]]:
            for leg2_start, leg2_end in [legs[1], legs[1][::-1]]:
                if leg1_start == leg2_start:
                    a = leg1_start
                    b = leg1_end
                    c = leg2_end
                    
                    # 1. Angle verification (0 < angle <= 90)
                    cross = cross_product(a, b, c)
                    dot = dot_product(a, b, c)
                    
                    if cross == 0 or dot < 0:
                        continue # Collinear or obtuse angle
                    
                    # 2. Bridge endpoints verification
                    p, q = bridge[0], bridge[1]
                    
                    # Case 1: P is on AB and Q is on AC
                    if is_on_segment_with_ratio(a, b, p) and is_on_segment_with_ratio(a, c, q):
                        return "YES"
                    # Case 2: Q is on AB and P is on AC
                    if is_on_segment_with_ratio(a, b, q) and is_on_segment_with_ratio(a, c, p):
                        return "YES"
                        
    return "NO"
 
def main():
    test_cases = read_segments()
    results = [solve_case(case) for case in test_cases]
    print('
'.join(results))
 
if __name__ == '__main__':
    main()
