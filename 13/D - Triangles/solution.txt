import sys
 
input = sys.stdin.readline
 
n, m = map(int, input().split())
 
red = [tuple(map(int, input().split())) for _ in range(n)]
blue = [tuple(map(int, input().split())) for _ in range(m)]
 
# Sort red points by y-coordinate
red.sort(key=lambda p: p[1])
 
f = [[0] * n for _ in range(n)]
 
for i in range(n):
    x1, y1 = red[i]
    for j in range(i + 1, n):
        x2, y2 = red[j]
        cnt = 0
        for xb, yb in blue:
            if yb <= y1 or yb > y2:
                continue
            if (x2 - x1) * (yb - y1) - (y2 - y1) * (xb - x1) > 0:
                cnt += 1
        f[i][j] = cnt
 
ans = 0
for i in range(n):
    for j in range(i + 1, n):
        for k in range(j + 1, n):
            if f[i][j] + f[j][k] == f[i][k]:
                ans += 1
 
print(ans)
