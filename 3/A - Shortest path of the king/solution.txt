s = input().strip()
t = input().strip()
 
x1, y1 = ord(s[0]) - ord('a'), int(s[1])
x2, y2 = ord(t[0]) - ord('a'), int(t[1])
 
moves = []
 
while x1 != x2 or y1 != y2:
    move = ""
 
    # Horizontal part first
    if x1 < x2:
        move += "R"
        x1 += 1
    elif x1 > x2:
        move += "L"
        x1 -= 1
 
    # Vertical part second
    if y1 < y2:
        move += "U"
        y1 += 1
    elif y1 > y2:
        move += "D"
        y1 -= 1
 
    moves.append(move)
 
print(len(moves))
print(*moves, sep='
')
